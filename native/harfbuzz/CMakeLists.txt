cmake_minimum_required(VERSION 3.16)
project(ExcelRendererHarfBuzzSubset LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(HB_BUILD_SUBSET ON CACHE BOOL "" FORCE)
set(HB_HAVE_FREETYPE OFF CACHE BOOL "" FORCE)
set(HB_HAVE_GLIB OFF CACHE BOOL "" FORCE)
set(HB_HAVE_ICU OFF CACHE BOOL "" FORCE)
add_subdirectory("${HB_SOURCE}" harfbuzz)
# Upstream static subset repeats support units already in the amalgamated core.
# Link each of those units once in the combined adapter.
get_target_property(SUBSET_SOURCES harfbuzz-subset SOURCES)
list(FILTER SUBSET_SOURCES EXCLUDE REGEX "(hb-static|hb-number|hb-ot-cff1-table|hb-ot-cff2-table)\\.cc$")
set_property(TARGET harfbuzz-subset PROPERTY SOURCES "${SUBSET_SOURCES}")
add_library(ExcelRenderer.HarfBuzzSubset SHARED subset.cpp)
target_link_libraries(ExcelRenderer.HarfBuzzSubset PRIVATE harfbuzz-subset harfbuzz)
target_include_directories(ExcelRenderer.HarfBuzzSubset PRIVATE "${HB_SOURCE}/src")
if(UNIX AND NOT APPLE)
  target_link_options(ExcelRenderer.HarfBuzzSubset PRIVATE "-Wl,--exclude-libs,ALL")
endif()
